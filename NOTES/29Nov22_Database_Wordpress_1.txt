---
- name: "Wordpress Site Deployment"
  hosts: wordpress
  become: true
  vars_files:
    - webserver.vars
  tasks:

    - name: "Install Packages"
      apt:
        name: "{{ packages }}"
        state: present
        update_cache: true
      register: package_status

    - name: "Creating Virtual Host Config file from template"
      template:
        src: nginx.reshmamjoby.in.tmpl
        dest: "/etc/nginx/sites-available/{{ domain_name }}.conf"
        owner: "{{ nginx_owner }}"
        group: "{{ nginx_group }}"
      register: vhosts_status

    - name: "Create symlink to WordPress site {{ domain_name }}"
      file:
        src: "/etc/nginx/sites-available/{{ domain_name }}.conf"
        dest: "/etc/nginx/sites-enabled/{{ domain_name }}.conf"
        state: link
        owner: "{{ nginx_owner }}"
        group: "{{ nginx_group }}"

    - name: "Remove default site"
      ignore_errors: true
      file:
        path: "/etc/nginx/sites-enabled/default"
        state: absent

    - name: "Create Document Root /var/www/html/{{ domain_name }}/"
      file:
        path: /var/www/html/{{ domain_name }}/
        state: directory
        owner: "{{ nginx_owner }}"
        group: "{{ nginx_group }}"

    - name: "Copying WordPress files"
      copy:
        src: "./wordpress/"
        dest: "/var/www/html/{{ domain_name }}/"
        owner: "{{ nginx_owner }}"
        group: "{{ nginx_group }}"

    - name: "MariaDB - Setting Root Password"
      ignore_errors: true
      mysql_user:
        login_user: "root"
        login_password: ""
        user: "root"
        password: "{{ mysql_root_password }}"
        host_all: true

    - name: "Removing Anonymous User"
      mysql_user:
        login_user: "root"
        login_password: "{{ mysql_root_password }}"
        user: ""
        host_all: true
        state: absent

    - name: "Removing Test Database"
      mysql_db:
        login_user: "root"
        login_password: "{{ mysql_root_password }}"
        name: "test"
        state: absent

    - name: "Creating WordPress Database"
      mysql_db:
        login_user: "root"
        login_password: "{{ mysql_root_password }}"
        name: "{{ wp_database }}"
        state: present

    - name: "Creating WordPress user and grant privileges"
      mysql_user:
        login_user: "root"
        login_password: "{{ mysql_root_password }}"
        user: "{{ wp_user }}"
        host: "%"
        password: "{{ wp_password }}"
        priv: '{{ wp_database }}.*:ALL'
        state: present
      register: database_status

    - name: "Restarting/Enabling nginx or mariadb service"
      when: package_status.changed == true or vhosts_status == true or database_status == true
      service:
        name: "{{ item }}"
        state: restarted
        enabled: true
      with_items: "{{ services }}"
