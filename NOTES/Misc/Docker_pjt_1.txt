/*Update the Nginx site configuration File for load balancing the Cachine Containers*/

vim /home/ec2-user/default.conf

upstream ajlb {
           server geolocation-api-service1:8080;
           server geolocation-api-service2:8080;
           server geolocation-api-service3:8080;
}
server {
        listen 443;
        listen [::]:443 ssl;

    ssl on;
    ssl_certificate             /etc/nginx/conf.d/nginx_reshmamjoby_in.crt;
    ssl_certificate_key     /etc/nginx/conf.d/nginx_reshmamjoby_in.key;

        # Add index.php to the list if you are using PHP
        index index.html index.htm index.php;

        server_name ipcache.reshmamjoby.in;

        location / {
                proxy_pass http://ajlb;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }
}
server {

    listen 80;
    listen [::]:80;
    server_name ipcache.reshmamjoby.in;
    return 301 https://$host$request_uri;
    location / {
                proxy_pass http://ajlb;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }

}

docker container run \
--name nginxlb \
-d \
-v /home/ec2-user/default.conf:/etc/nginx/conf.d/default.conf \
-v /home/ec2-user/nginx_reshmamjoby_in.crt:/etc/nginx/conf.d/nginx_reshmamjoby_in.crt \
-v /home/ec2-user/nginx_reshmamjoby_in.key:/etc/nginx/conf.d/nginx_reshmamjoby_in.key \
-p 80:80 \
-p 443:443 \
nginx:latest
