#!/bin/bash
 
echo "ClientAliveInterval 60" >> /etc/ssh/sshd_config
echo "LANG=en_US.utf-8" >> /etc/environment
echo "LC_ALL=en_US.utf-8" >> /etc/environment
sudo yum install sshd -y
sudo service sshd restart

sudo hostnamectl set-hostname aws-infra

yum install httpd -y
systemctl restart httpd.service
systemctl enable httpd.service

---
- name: "Amazon Infra Creation"
  hosts: localhost
  vars_files:
    - main.vars
  tasks:
    - name: "Creating Ec2 Instance"
      amazon.aws.ec2_instance:
        aws_access_key: "{{ access_key }}"
        aws_secret_key: "{{ secret_key }}"
        region: "{{ region }}"
        key_name: "{{ instance_key }}"
        instance_type: "{{ instance_type }}"
        image_id: "{{ instance_ami }}"
        tags: "{{ instance_tags }}"
        security_groups: "{{ group }}"
        exact_count: 2
        filters:
          "tag:Name" : "Learn-new"
          "instance-state-name": "running"


    - name: "Fetching Ec2 Instance Details"
      amazon.aws.ec2_instance_info:
        aws_access_key: "{{ access_key }}"
        aws_secret_key: "{{ secret_key }}"
        region: "{{ region }}"
        filters:
          "tag:Name" : "Learn-new"
          "instance-state-name": [ "running"]
      register: ec2info

    - debug:
        var: ec2info
==========================================

    - debug:
        msg: "Instance ID: {{ item.instance_id }} Public IP: {{ item.public_ip_address }}"
      with_items: "{{ ec2info.instances }}"

==========================================

    - name: "Creating Dynamic Inventory"
      add_host:
        
        hostname: "{{ item.public_ip_address }}"
        ansible_ssh_user: "ec2-user"
        ansible_ssh_host: "{{ item.public_ip_address }}"
        ansible_ssh_port: 22
        groups:
          - amazon
        ansible_ssh_private_key_file: "ssh.pem"
        ansible_ssh_common_args: "-o StrictHostKeyChecking=no"
            
      with_items: "{{ ec2.instances }}"

- name: "Installing apache on amazon group"
  become: true
  hosts: amazon
  tasks:
    
    - name: "Installing apache"
      yum:
        name: "httpd"
        state: present
            
    - name: "creating index.html"
      copy:
        content: "<center><h1> {{ ansible_fqdn }} </h1></center>"
        dest: /var/www/html/index.html
            
    - name: "restarting"
      service:
        name: httpd
        state: restarted
        enabled: true
